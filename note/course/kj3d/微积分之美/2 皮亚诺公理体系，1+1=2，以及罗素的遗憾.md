from sympy import isprime

isprime(31)  # True
isprime(331)  # True
isprime(3331)  # True
isprime(33331)  # True
isprime(333331)  # True
isprime(3333331)  # True
isprime(33333331)  # True
# 333...1 prime
isprime(333333331)  # False ⚠️
print(333333331 / 17)  # 19607843.0
